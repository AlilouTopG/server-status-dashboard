    <script>
        let isSoundMuted = false;
        let currentFilter = 'all';
        let webhookUrl = localStorage.getItem('monitoring_webhook') || '';

        // قائمة السيرفرات مع إحداثيات GPS حقيقية
        let servers = [
            { id: 1, name: "CDN Global Gateway", region: "Frankfurt", env: "Production", lat: 50.1109, lng: 8.6821, status: "operational", cpu: 18, latency: 12 },
            { id: 2, name: "Auth Cluster US", region: "Virginia", env: "Production", lat: 37.4316, lng: -78.6569, status: "operational", cpu: 32, latency: 45 },
            { id: 3, name: "Primary Postgres Database", region: "Ireland", env: "Database", lat: 53.3498, lng: -6.2603, status: "operational", cpu: 40, latency: 28 },
            { id: 4, name: "Staging Testing Server", region: "Singapore", env: "Staging", lat: 1.3521, lng: 103.8198, status: "operational", cpu: 15, latency: 85 }
        ];

        let map;
        let mapMarkers = {};

        function initMap() {
            // إنشاء الخريطة وتحديد المركز العالمي
            map = L.map('worldMap').setView([20, 0], 2);
            L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
                maxZoom: 18,
                attribution: '© OpenStreetMap'
            }).addTo(map);

            updateMapMarkers();
        }

        function updateMapMarkers() {
            // مسح العلامات القديمة وإعادة رسم السيرفرات
            Object.values(mapMarkers).forEach(marker => map.removeLayer(marker));
            mapMarkers = {};

            servers.forEach(s => {
                const markerColor = s.status === 'down' ? '#ef4444' : '#10b981';
                const customIcon = L.divIcon({
                    className: 'custom-map-pin',
                    html: `<div style="background-color: ${markerColor}; width: 14px; height: 14px; border-radius: 50%; border: 2px solid white; box-shadow: 0 0 10px ${markerColor};"></div>`
                });

                const marker = L.marker([s.lat, s.lng], { icon: customIcon }).addTo(map);
                marker.bindPopup(`<b>${s.name}</b><br>${s.region}<br>الحالة: ${s.status === 'down' ? 'عطل 🚨' : 'شغال ✅'}`);
                mapMarkers[s.id] = marker;
            });
        }

        // --------------------------------------------------
        // نظام الصوت المطور
        // --------------------------------------------------
        let audioCtx = null;
        function getAudioContext() {
            if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            if (audioCtx.state === 'suspended') audioCtx.resume();
            return audioCtx;
        }

        function playBeep(freq, type, duration) {
            if (isSoundMuted) return;
            try {
                const ctx = getAudioContext();
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                osc.type = type;
                osc.frequency.value = freq;
                osc.connect(gain);
                gain.connect(ctx.destination);
                osc.start();
                gain.gain.setValueAtTime(0.1, ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.00001, ctx.currentTime + duration);
                osc.stop(ctx.currentTime + duration);
            } catch (e) {}
        }

        function playAlertSound() { 
            playBeep(440, 'sawtooth', 0.2); 
            setTimeout(() => playBeep(330, 'sawtooth', 0.4), 200);
        }

        function playSuccessSound() { 
            playBeep(523.25, 'sine', 0.15); 
            setTimeout(() => playBeep(659.25, 'sine', 0.25), 150);
        }

        function toggleSound() {
            isSoundMuted = !isSoundMuted;
            getAudioContext();
            document.getElementById('soundBtn').innerHTML = isSoundMuted 
                ? `<i class="fa-solid fa-volume-xmark text-red-500"></i>`
                : `<i class="fa-solid fa-volume-high text-indigo-500"></i>`;
        }

        // Webhook integration (Discord Notification)
        function sendWebhookNotification(serverName, reason) {
            if (!webhookUrl) return;
            const payload = {
                content: `🚨 **تنبيه عطل طارئ!**\n**السيرفر المتأثر:** ${serverName}\n**السبب:** ${reason}\n**الوقت:** ${new Date().toLocaleTimeString()}`
            };
            fetch(webhookUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            }).catch(err => console.error("فشل إرسال Webhook:", err));
        }

        // Chart configuration
        let telemetryChart;
        function initChart() {
            const ctx = document.getElementById('telemetryChart').getContext('2d');
            const isDark = document.documentElement.classList.contains('dark');
            const textColor = isDark ? '#94a3b8' : '#64748b';

            telemetryChart = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: ['15s', '12s', '9s', '6s', '3s', 'الآن'],
                    datasets: [
                        { label: 'متوسط CPU (%)', data: [20, 25, 22, 30, 28, 24], borderColor: '#10b981', backgroundColor: 'rgba(16, 185, 129, 0.08)', fill: true, tension: 0.4 },
                        { label: 'معدل Latency (ms)', data: [25, 28, 26, 32, 30, 27], borderColor: '#6366f1', backgroundColor: 'rgba(99, 102, 241, 0.08)', fill: true, tension: 0.4 }
                    ]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { labels: { color: textColor } } },
                    scales: {
                        x: { ticks: { color: textColor }, grid: { color: 'rgba(150,150,150,0.1)' } },
                        y: { ticks: { color: textColor }, grid: { color: 'rgba(150,150,150,0.1)' }, min: 0 }
                    }
                }
            });
        }

        function toggleTheme() {
            const html = document.documentElement;
            const icon = document.getElementById('themeIcon');
            if (html.classList.contains('dark')) {
                html.classList.remove('dark');
                icon.className = 'fa-solid fa-sun text-amber-500';
            } else {
                html.classList.add('dark');
                icon.className = 'fa-solid fa-moon text-amber-400';
            }
            if (telemetryChart) {
                const isDark = html.classList.contains('dark');
                const textColor = isDark ? '#94a3b8' : '#64748b';
                telemetryChart.options.plugins.legend.labels.color = textColor;
                telemetryChart.options.scales.x.ticks.color = textColor;
                telemetryChart.options.scales.y.ticks.color = textColor;
                telemetryChart.update();
            }
        }

        function setFilter(filter) {
            currentFilter = filter;
            renderServers();
        }

        function applyFilters() { renderServers(); }

        function renderServers() {
            const grid = document.getElementById('serversGrid');
            const searchVal = document.getElementById('searchInput').value.toLowerCase();
            const envVal = document.getElementById('envFilter').value;

            let filtered = servers.filter(s => {
                const matchesSearch = s.name.toLowerCase().includes(searchVal) || s.region.toLowerCase().includes(searchVal);
                const matchesEnv = envVal === 'all' || s.env === envVal;
                const matchesFilter = currentFilter === 'all' || s.status === currentFilter;
                return matchesSearch && matchesEnv && matchesFilter;
            });

            grid.innerHTML = '';
            filtered.forEach(s => {
                const isDown = s.status === 'down';
                let statusColor = isDown ? 'bg-red-500 animate-ping' : 'bg-emerald-500';
                let statusBadge = isDown ? 'bg-red-500/10 text-red-500 border-red-500/20' : 'bg-emerald-500/10 text-emerald-500 border-emerald-500/20';

                grid.innerHTML += `
                    <div class="glass-panel p-4 rounded-xl space-y-3 relative overflow-hidden border-t-2 ${isDown ? 'border-t-red-500' : 'border-t-emerald-500'}">
                        <div class="flex justify-between items-start">
                            <div>
                                <div class="font-bold text-xs flex items-center gap-1.5">
                                    <span class="w-2 h-2 rounded-full ${statusColor}"></span>
                                    ${s.name}
                                </div>
                                <div class="text-[10px] text-muted mt-0.5">${s.region} • <span class="text-indigo-400 font-semibold">${s.env}</span></div>
                            </div>
                            <span class="text-[9px] px-1.5 py-0.5 rounded border ${statusBadge}">${isDown ? 'DOWN' : 'OK'}</span>
                        </div>

                        <div class="flex justify-between items-center text-xs pt-2 border-t border-slate-700/20">
                            <div>CPU: <span class="font-semibold">${s.cpu}%</span></div>
                            <div>Latency: <span class="font-semibold">${s.latency}ms</span></div>
                            <button onclick="fixServer(${s.id})" class="text-[10px] bg-indigo-600/20 hover:bg-indigo-600/40 text-indigo-500 border border-indigo-500/30 px-2 py-1 rounded-lg transition">
                                <i class="fa-solid fa-wrench"></i> ${isDown ? 'إصلاح' : 'إعادة تشغيل'}
                            </button>
                        </div>
                    </div>
                `;
            });

            updateMapMarkers();
        }

        // Dynamic Loop
        setInterval(() => {
            servers.forEach(s => {
                if(s.status !== 'down') {
                    s.cpu = Math.min(100, Math.max(10, s.cpu + Math.floor(Math.random() * 7 - 3)));
                    s.latency = Math.min(150, Math.max(8, s.latency + Math.floor(Math.random() * 5 - 2)));
                }
            });
            renderServers();

            if (telemetryChart) {
                const avgCpu = Math.floor(servers.reduce((a,b) => a + b.cpu, 0) / servers.length);
                telemetryChart.data.datasets[0].data.shift();
                telemetryChart.data.datasets[0].data.push(avgCpu);
                telemetryChart.update();
            }

            document.getElementById('avgLatency').innerText = Math.floor(servers.reduce((a,b)=>a+b.latency,0)/servers.length) + ' ms';
        }, 3000);

        function triggerGlobalIncident() {
            const target = servers[0];
            if(target) {
                target.status = 'down';
                target.cpu = 99;
                target.latency = 500;
                sendWebhookNotification(target.name, "ارتفاع مفاجئ في استهلاك CPU وسقوط الاتصال.");
            }
            playAlertSound();
            renderServers();

            const log = document.getElementById('logsContainer');
            log.innerHTML = `
                <div class="p-2 rounded-lg border border-red-500/30 bg-red-500/10 text-xs">
                    🚨 <b>عطل طارئ:</b> ${target.name} متوقف حالياً. تم إرسال تنبيه Webhook.
                </div>
            ` + log.innerHTML;
        }

        function fixServer(id) {
            const server = servers.find(s => s.id === id);
            if(server) {
                server.status = 'operational';
                server.cpu = 20;
                server.latency = 15;
                playSuccessSound();
                renderServers();

                const log = document.getElementById('logsContainer');
                log.innerHTML = `
                    <div class="p-2 rounded-lg border border-emerald-500/30 bg-emerald-500/10 text-xs">
                        ✅ <b>تم الإصلاح:</b> استقرار السيرفر ${server.name}.
                    </div>
                ` + log.innerHTML;
            }
        }

        // Webhook Modal logic
        function openWebhookModal() { 
            document.getElementById('webhookUrlInput').value = webhookUrl;
            document.getElementById('webhookModal').classList.remove('hidden'); 
        }
        function closeWebhookModal() { document.getElementById('webhookModal').classList.add('hidden'); }
        function saveWebhook() {
            webhookUrl = document.getElementById('webhookUrlInput').value;
            localStorage.setItem('monitoring_webhook', webhookUrl);
            document.getElementById('webhookStatusDisplay').innerText = webhookUrl ? "مُفعل ومتصل ✅" : "غير متصل";
            closeWebhookModal();
        }

        function openAddServerModal() { document.getElementById('addServerModal').classList.remove('hidden'); }
        function closeAddServerModal() { document.getElementById('addServerModal').classList.add('hidden'); }
        function saveNewServer() {
            const name = document.getElementById('newServerName').value;
            const region = document.getElementById('newServerRegion').value;
            const cords = document.getElementById('newServerCords').value.split(',');
            const env = document.getElementById('newServerEnv').value;

            if(name && region && cords.length === 2) {
                servers.push({
                    id: Date.now(),
                    name, region, env,
                    lat: parseFloat(cords[0]),
                    lng: parseFloat(cords[1]),
                    status: 'operational', cpu: 15, latency: 20
                });
                renderServers();
                closeAddServerModal();
            }
        }

        function exportReport() {
            const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(servers, null, 2));
            const anchor = document.createElement('a');
            anchor.setAttribute("href", dataStr);
            anchor.setAttribute("download", `cloud-report-${Date.now()}.json`);
            anchor.click();
        }
